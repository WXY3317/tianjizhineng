cmake_minimum_required(VERSION 3.16)
project(gripper_RS485 LANGUAGES CXX)
include(CTest)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../ros2_ws/src/gripper/example" demo/example)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Eigen3 REQUIRED)
string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" calibration_processor)
if(calibration_processor MATCHES "^(aarch64|arm64)$")
    set(calibration_sdk_lib "${CMAKE_CURRENT_SOURCE_DIR}/lib/arm")
elseif(calibration_processor MATCHES "^(x86_64|amd64)$")
    set(calibration_sdk_lib "${CMAKE_CURRENT_SOURCE_DIR}/lib/x86")
else()
    message(FATAL_ERROR "Unsupported processor: ${CMAKE_SYSTEM_PROCESSOR}")
endif()
add_executable(gripper_calibrate src/calibrate_gripper.cpp)
target_include_directories(gripper_calibrate PRIVATE include)
target_link_libraries(gripper_calibrate PRIVATE Eigen3::Eigen
    "-Wl,--no-as-needed" "${calibration_sdk_lib}/libgripper_lib.so"
    "${calibration_sdk_lib}/libgrip_pawl_lib.so" "${calibration_sdk_lib}/libcan_motor_driver.so")
set_target_properties(gripper_calibrate PROPERTIES BUILD_RPATH "${calibration_sdk_lib}")

option(GRIPPER_BUILD_EXAMPLES "Build vendor motion examples" OFF)
if(GRIPPER_BUILD_EXAMPLES)
    set(CMAKE_CXX_STANDARD 17)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    find_package(Eigen3 REQUIRED)
    string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" processor)
    if(processor MATCHES "^(aarch64|arm64)$")
        set(sdk_lib "${CMAKE_CURRENT_SOURCE_DIR}/lib/arm")
    elseif(processor MATCHES "^(x86_64|amd64)$")
        set(sdk_lib "${CMAKE_CURRENT_SOURCE_DIR}/lib/x86")
    else()
        message(FATAL_ERROR "Unsupported processor: ${CMAKE_SYSTEM_PROCESSOR}")
    endif()
    add_library(gripper_tools_lib STATIC src/gripper_tools.cpp)
    target_include_directories(gripper_tools_lib PUBLIC include)
    target_link_libraries(gripper_tools_lib PUBLIC Eigen3::Eigen
        "-Wl,--no-as-needed" "${sdk_lib}/libgripper_lib.so"
        "${sdk_lib}/libgrip_pawl_lib.so" "${sdk_lib}/libcan_motor_driver.so")
    foreach(example position_loop_demo velocity_loop_demo torque_loop_demo zero_demo)
        add_executable(${example} src/${example}.cpp)
        target_link_libraries(${example} PRIVATE gripper_tools_lib)
        set_target_properties(${example} PROPERTIES BUILD_RPATH "${sdk_lib}")
    endforeach()
endif()
