cmake_minimum_required(VERSION 3.16)
project(gripper_message_demo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Eigen3 REQUIRED)
find_package(Threads REQUIRED)
set(SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." CACHE PATH "SDK root containing include/ and lib/")
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
    set(SDK_LIB_SUBDIR "arm")
else()
    set(SDK_LIB_SUBDIR "x86")
endif()
set(SDK_LIB "${SDK_ROOT}/lib/${SDK_LIB_SUBDIR}")
foreach(SDK_SO libgripper_lib.so libgrip_pawl_lib.so libcan_motor_driver.so)
    if(NOT EXISTS "${SDK_LIB}/${SDK_SO}")
        message(FATAL_ERROR "Missing SDK library: ${SDK_LIB}/${SDK_SO}")
    endif()
endforeach()
add_library(gripper_core controller.cpp)
target_include_directories(gripper_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(gripper_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(gripper_demo main.cpp hardware.cpp)
target_include_directories(gripper_demo PRIVATE "${SDK_ROOT}/include")
target_compile_options(gripper_demo PRIVATE -Wall -Wextra -Wpedantic)
configure_file(calibration.csv "${CMAKE_CURRENT_BINARY_DIR}/calibration.csv" COPYONLY)
configure_file(right_calibration.csv "${CMAKE_CURRENT_BINARY_DIR}/right_calibration.csv" COPYONLY)
target_link_libraries(gripper_demo PRIVATE gripper_core Eigen3::Eigen Threads::Threads
    "-Wl,--no-as-needed" "${SDK_LIB}/libgripper_lib.so"
    "${SDK_LIB}/libgrip_pawl_lib.so" "${SDK_LIB}/libcan_motor_driver.so")
set_target_properties(gripper_demo PROPERTIES BUILD_RPATH "\$ORIGIN/../lib/${SDK_LIB_SUBDIR};${SDK_LIB}")
add_executable(gripper_inspect inspect_position.cpp)
target_include_directories(gripper_inspect PRIVATE "${SDK_ROOT}/include")
target_link_libraries(gripper_inspect PRIVATE Eigen3::Eigen Threads::Threads
    "-Wl,--no-as-needed" "${SDK_LIB}/libgripper_lib.so"
    "${SDK_LIB}/libgrip_pawl_lib.so" "${SDK_LIB}/libcan_motor_driver.so")
set_target_properties(gripper_inspect PROPERTIES BUILD_RPATH "\$ORIGIN/../lib/${SDK_LIB_SUBDIR};${SDK_LIB}")
include(CTest)
if(BUILD_TESTING)
    add_executable(controller_test tests.cpp)
    target_link_libraries(controller_test PRIVATE gripper_core)
    target_compile_options(controller_test PRIVATE -Wall -Wextra -Wpedantic)
    add_test(NAME controller_state_machine COMMAND controller_test)
    add_executable(calibration_test calibration_test.cpp)
    target_include_directories(calibration_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_definitions(calibration_test PRIVATE DEMO_CALIBRATION_PATH="${CMAKE_CURRENT_SOURCE_DIR}/calibration.csv")
    target_compile_definitions(calibration_test PRIVATE EXPECTED_OPEN_WIDTH=0.098 EXPECTED_CLOSED_ANGLE=61.2092854977)
    add_test(NAME calibration_mapping COMMAND calibration_test)
    add_executable(right_calibration_test calibration_test.cpp)
    target_include_directories(right_calibration_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_definitions(right_calibration_test PRIVATE DEMO_CALIBRATION_PATH="${CMAKE_CURRENT_SOURCE_DIR}/right_calibration.csv"
        EXPECTED_OPEN_WIDTH=0.098 EXPECTED_CLOSED_ANGLE=61.2092854977
        EXPECTED_MEASURED_ANGLE=29.6966209412 EXPECTED_MEASURED_WIDTH=0.057)
    add_test(NAME right_calibration_mapping COMMAND right_calibration_test)
endif()
