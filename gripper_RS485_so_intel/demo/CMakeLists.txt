cmake_minimum_required(VERSION 3.16)
project(gripper_message_demo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Eigen3 REQUIRED)
find_package(Threads REQUIRED)
set(SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
set(SDK_LIB "${SDK_ROOT}/lib/x86")
add_library(gripper_core controller.cpp)
target_include_directories(gripper_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(gripper_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(gripper_demo main.cpp hardware.cpp)
target_include_directories(gripper_demo PRIVATE "${SDK_ROOT}/include")
target_compile_options(gripper_demo PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(gripper_demo PRIVATE DEMO_CALIBRATION_PATH="${CMAKE_CURRENT_SOURCE_DIR}/calibration.csv")
target_link_libraries(gripper_demo PRIVATE gripper_core Eigen3::Eigen Threads::Threads
    "-Wl,--no-as-needed" "${SDK_LIB}/libgripper_lib.so"
    "${SDK_LIB}/libgrip_pawl_lib.so" "${SDK_LIB}/libcan_motor_driver.so")
set_target_properties(gripper_demo PROPERTIES BUILD_RPATH "${SDK_LIB}")
add_executable(gripper_inspect inspect_position.cpp)
target_include_directories(gripper_inspect PRIVATE "${SDK_ROOT}/include")
target_link_libraries(gripper_inspect PRIVATE Eigen3::Eigen Threads::Threads
    "-Wl,--no-as-needed" "${SDK_LIB}/libgripper_lib.so"
    "${SDK_LIB}/libgrip_pawl_lib.so" "${SDK_LIB}/libcan_motor_driver.so")
set_target_properties(gripper_inspect PROPERTIES BUILD_RPATH "${SDK_LIB}")
include(CTest)
if(BUILD_TESTING)
    add_executable(controller_test tests.cpp)
    target_link_libraries(controller_test PRIVATE gripper_core)
    target_compile_options(controller_test PRIVATE -Wall -Wextra -Wpedantic)
    add_test(NAME controller_state_machine COMMAND controller_test)
    add_executable(calibration_test calibration_test.cpp)
    target_include_directories(calibration_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_definitions(calibration_test PRIVATE DEMO_CALIBRATION_PATH="${CMAKE_CURRENT_SOURCE_DIR}/calibration.csv")
    add_test(NAME calibration_mapping COMMAND calibration_test)
endif()
