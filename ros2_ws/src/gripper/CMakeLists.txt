cmake_minimum_required(VERSION 3.16)
project(gripper LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(builtin_interfaces REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(Eigen3 REQUIRED)
get_filename_component(PACKAGE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}" REALPATH)
set(SDK_ROOT "${PACKAGE_ROOT}/sdk")
string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" processor)
if(processor MATCHES "^(aarch64|arm64)$")
  set(SDK_LIB "${SDK_ROOT}/lib/arm")
elseif(processor MATCHES "^(x86_64|amd64)$")
  set(SDK_LIB "${SDK_ROOT}/lib/x86")
else()
  message(FATAL_ERROR "Unsupported processor: ${CMAKE_SYSTEM_PROCESSOR}")
endif()
rosidl_generate_interfaces(${PROJECT_NAME}
  "msg/GripperCommand.msg" "msg/GripperState.msg" "msg/GripperAck.msg"
  DEPENDENCIES builtin_interfaces)
add_executable(gripper_node src/gripper_node.cpp src/controller.cpp src/hardware.cpp)
target_include_directories(gripper_node PRIVATE src "${SDK_ROOT}/include")
target_compile_definitions(gripper_node PRIVATE DEMO_CALIBRATION_PATH="")
ament_target_dependencies(gripper_node rclcpp rcl_interfaces ament_index_cpp)
target_link_libraries(gripper_node Eigen3::Eigen "-Wl,--no-as-needed"
  "${SDK_LIB}/libgripper_lib.so" "${SDK_LIB}/libgrip_pawl_lib.so"
  "${SDK_LIB}/libcan_motor_driver.so")
rosidl_get_typesupport_target(typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(gripper_node "${typesupport}")
set_target_properties(gripper_node PROPERTIES INSTALL_RPATH "$ORIGIN")
add_executable(gripper_open src/gripper_open.cpp)
target_include_directories(gripper_open PRIVATE "${SDK_ROOT}/include")
target_compile_options(gripper_open PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(gripper_open Eigen3::Eigen "-Wl,--no-as-needed"
  "${SDK_LIB}/libgripper_lib.so" "${SDK_LIB}/libgrip_pawl_lib.so"
  "${SDK_LIB}/libcan_motor_driver.so")
set_target_properties(gripper_open PROPERTIES INSTALL_RPATH "$ORIGIN")
install(TARGETS gripper_node gripper_open DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY "${SDK_LIB}/" DESTINATION lib/${PROJECT_NAME} FILES_MATCHING PATTERN "*.so")
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(controller_test example/tests.cpp src/controller.cpp)
  target_include_directories(controller_test PRIVATE src)
  add_test(NAME controller_state_machine COMMAND controller_test)
  add_executable(calibration_test example/calibration_test.cpp)
  target_include_directories(calibration_test PRIVATE src)
  target_compile_definitions(calibration_test PRIVATE DEMO_CALIBRATION_PATH="${CMAKE_CURRENT_SOURCE_DIR}/config/calibration.csv")
  add_test(NAME calibration_mapping COMMAND calibration_test)
endif()
ament_export_dependencies(rosidl_default_runtime)
ament_package()
